# Plan 02-01: AI Handler

## Goal

Xây dựng Brain Handler kết nối với Gemini hoặc Groq API. Handler nhận context (tên app + loại app) và trả về JSON chứa status, message, energy_change để điều khiển hành vi của bé Slime.

### Task 2: Xây dựng BrainHandler (Groq + Gemini)

<action>
Tạo `src/brain/__init__.py` rỗng và file `src/brain/brain_handler.py`:

```python
import json
import os
import time
from brain.prompts import SYSTEM_PROMPT, USER_PROMPT_TEMPLATE

# ---- Response Data Class ----

class SlimeResponse:
    """Kết quả phản hồi từ AI đã được parse."""
    def __init__(self, status: str, message: str, energy_change: int):
        self.status = status            # Happy | Sad | Angry | Evolving
        self.message = message          # Câu nói của bé Slime
        self.energy_change = energy_change  # -10 đến +10

    @classmethod
    def default(cls) -> "SlimeResponse":
        """Phản hồi mặc định khi AI lỗi."""
        return cls(status="Sad", message="Tôi đang ngủ gật một chút...", energy_change=0)


# ---- Brain Handler ----

class BrainHandler:
    """
    Kết nối với Groq hoặc Gemini API để tạo phản hồi cho bé Slime.
    Ưu tiên Groq (nhanh hơn), fallback sang Gemini nếu Groq lỗi.
    """

    MIN_CALL_INTERVAL = 30  # Tối thiểu 30 giây giữa các lần gọi API

    def __init__(self, groq_api_key: str = None, gemini_api_key: str = None):
        self._last_call_time = 0
        self._groq_client = None
        self._gemini_model = None

        # Khởi tạo Groq client
        groq_key = groq_api_key or os.getenv("GROQ_API_KEY", "")
        if groq_key:
            try:
                from groq import Groq
                self._groq_client = Groq(api_key=groq_key)
                print("[Brain] Groq client sẵn sàng.")
            except ImportError:
                print("[Brain] Groq không được cài. Chạy: pip install groq")

        # Khởi tạo Gemini client (fallback)
        gemini_key = gemini_api_key or os.getenv("GEMINI_API_KEY", "")
        if gemini_key:
            try:
                import google.generativeai as genai
                genai.configure(api_key=gemini_key)
                self._gemini_model = genai.GenerativeModel(
                    model_name="gemini-2.0-flash",
                    system_instruction=SYSTEM_PROMPT
                )
                print("[Brain] Gemini client sẵn sàng.")
            except ImportError:
                print("[Brain] google-generativeai không được cài. Chạy: pip install google-generativeai")

    def analyze(self, process_name: str, app_type: str,
                window_title: str, minutes_used: float = 0) -> SlimeResponse:
        """
        Phân tích ngữ cảnh và trả về phản hồi của bé Slime.
        Có rate limiting để tránh spam API.
        """
        # Rate limiting
        now = time.time()
        if now - self._last_call_time < self.MIN_CALL_INTERVAL:
            return SlimeResponse.default()
        self._last_call_time = now

        user_message = USER_PROMPT_TEMPLATE.format(
            process_name=process_name,
            app_type=app_type,
            window_title=window_title[:100],
            minutes_used=round(minutes_used, 1)
        )

        # Thử Groq trước
        if self._groq_client:
            response = self._call_groq(user_message)
            if response:
                return response

        # Fallback sang Gemini
        if self._gemini_model:
            response = self._call_gemini(user_message)
            if response:
                return response

        print("[Brain] Không có API nào khả dụng.")
        return SlimeResponse.default()

    def _call_groq(self, user_message: str) -> SlimeResponse | None:
        """Gọi Groq API (llama-3.1-8b-instant — free tier)."""
        try:
            completion = self._groq_client.chat.completions.create(
                model="llama-3.1-8b-instant",
                messages=[
                    {"role": "system", "content": SYSTEM_PROMPT},
                    {"role": "user", "content": user_message}
                ],
                temperature=0.8,
                max_tokens=150,
                response_format={"type": "json_object"}
            )
            raw = completion.choices[0].message.content
            return self._parse_response(raw)
        except Exception as e:
            print(f"[Brain] Groq lỗi: {e}")
            return None

    def _call_gemini(self, user_message: str) -> SlimeResponse | None:
        """Gọi Gemini API (gemini-2.0-flash — free tier)."""
        try:
            result = self._gemini_model.generate_content(user_message)
            raw = result.text
            return self._parse_response(raw)
        except Exception as e:
            print(f"[Brain] Gemini lỗi: {e}")
            return None

    def _parse_response(self, raw: str) -> SlimeResponse | None:
        """Parse JSON response từ AI."""
        try:
            # Làm sạch markdown code blocks nếu có
            raw = raw.strip()
            if raw.startswith("```"):
                raw = raw.split("```")[1]
                if raw.startswith("json"):
                    raw = raw[4:]

            data = json.loads(raw)
            status = data.get("status", "Sad")
            message = data.get("message", "...")
            energy_change = int(data.get("energy_change", 0))

            # Validate
            if status not in ("Happy", "Sad", "Angry", "Evolving"):
                status = "Sad"
            energy_change = max(-10, min(10, energy_change))

            return SlimeResponse(status=status, message=message, energy_change=energy_change)
        except (json.JSONDecodeError, KeyError, ValueError) as e:
            print(f"[Brain] Parse lỗi: {e} | Raw: {raw[:100]}")
            return None
```

### Task 3: Kết nối BrainHandler vào main.py

<action>
Cập nhật `src/main.py` để BrainHandler xử lý tín hiệu từ MonitorEngine:

```python
import sys
import os

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from PyQt6.QtWidgets import QApplication
from ui.pet_window import PetWindow
from monitor.monitor_engine import MonitorEngine
from brain.brain_handler import BrainHandler


def main():
    app = QApplication(sys.argv)
    app.setQuitOnLastWindowClosed(False)

    window = PetWindow()

    # Khởi tạo Brain (đọc API key từ env)
    brain = BrainHandler(
        groq_api_key=os.getenv("GROQ_API_KEY", ""),
        gemini_api_key=os.getenv("GEMINI_API_KEY", "")
    )

    monitor = MonitorEngine()

    def on_status_changed(process_name: str, app_type: str, window_title: str):
        print(f"[UI Update] {process_name} → {app_type}")
        # Gọi AI để tạo phản hồi
        response = brain.analyze(process_name, app_type, window_title)
        # Cập nhật UI (Phase 2 - Plan 02-02)
        window.update_mood(response.status, response.message)
        print(f"[Slime] {response.status} | {response.message} | Energy: {response.energy_change:+d}")

    monitor.status_changed.connect(on_status_changed)
    monitor.start()

    window.show()
    app.aboutToQuit.connect(monitor.stop)

    sys.exit(app.exec())


if __name__ == "__main__":
    main()
```
