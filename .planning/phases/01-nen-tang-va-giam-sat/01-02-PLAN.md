### Task 2: Xây dựng MonitorEngine (Background Thread)

<action>
Tạo file `src/monitor/__init__.py` rỗng và file `src/monitor/monitor_engine.py`:

```python
import json
import time
import threading
import os
import psutil
import win32gui
import win32process
from PyQt6.QtCore import QObject, pyqtSignal


class MonitorEngine(QObject):
    """
    Luồng nền giám sát cửa sổ foreground.
    Chạy mỗi POLL_INTERVAL giây, phát tín hiệu `status_changed`
    khi phát hiện ngữ cảnh mới.
    """

    # Tín hiệu gửi về UI: (tên_process, loại_app, tên_cửa_sổ)
    status_changed = pyqtSignal(str, str, str)

    POLL_INTERVAL = 10  # giây

    def __init__(self, mapping_path: str = None):
        super().__init__()
        self._running = False
        self._thread = None
        self._last_process = ""

        # Tải local mapping
        if mapping_path is None:
            base_dir = os.path.dirname(os.path.abspath(__file__))
            mapping_path = os.path.join(base_dir, "app_mapping.json")

        with open(mapping_path, "r", encoding="utf-8") as f:
            self._mapping = json.load(f)

        # Tạo flat lookup dict để tra cứu nhanh O(1)
        self._app_lookup: dict[str, str] = {}
        for category, apps in self._mapping.items():
            for app in apps:
                self._app_lookup[app.lower()] = category

    # ---- Public API ----

    def start(self):
        """Bắt đầu luồng giám sát."""
        self._running = True
        self._thread = threading.Thread(
            target=self._poll_loop,
            daemon=True,
            name="PomoSlime-Monitor"
        )
        self._thread.start()

    def stop(self):
        """Dừng luồng giám sát."""
        self._running = False

    # ---- Internal ----

    def _get_foreground_process(self) -> tuple[str, str]:
        """Lấy tên process và tên cửa sổ đang foreground."""
        try:
            hwnd = win32gui.GetForegroundWindow()
            window_title = win32gui.GetWindowText(hwnd)
            _, pid = win32process.GetWindowThreadProcessId(hwnd)
            process = psutil.Process(pid)
            return process.name(), window_title
        except Exception:
            return "", ""

    def _classify(self, process_name: str) -> str:
        """
        Phân loại ứng dụng.
        Trả về: 'Work', 'Distraction', 'Mixed', hoặc 'Unknown'.
        """
        return self._app_lookup.get(process_name.lower(), "Unknown")

    def _poll_loop(self):
        """Vòng lặp polling chính chạy trên background thread."""
        while self._running:
            process_name, window_title = self._get_foreground_process()

            # Chỉ cập nhật khi process thay đổi (tránh spam tín hiệu)
            if process_name and process_name != self._last_process:
                self._last_process = process_name
                app_type = self._classify(process_name)
                print(f"[Monitor] {process_name} → {app_type} | {window_title}")
                self.status_changed.emit(process_name, app_type, window_title)

            time.sleep(self.POLL_INTERVAL)
```

### Task 3: Kết nối MonitorEngine với PetWindow

<action>
Cập nhật `src/main.py` để khởi động MonitorEngine và kết nối tín hiệu với UI:

```python
import sys
from PyQt6.QtWidgets import QApplication
from ui.pet_window import PetWindow
from monitor.monitor_engine import MonitorEngine


def main():
    """Entry point của PomoSlime."""
    app = QApplication(sys.argv)
    app.setQuitOnLastWindowClosed(False)

    # Khởi tạo UI
    window = PetWindow()

    # Khởi tạo Monitor Engine
    monitor = MonitorEngine()

    # Kết nối tín hiệu: khi monitor phát hiện app mới → log ra console
    # (Phase 2 sẽ thay bằng gọi AI)
    def on_status_changed(process_name: str, app_type: str, window_title: str):
        print(f"[UI Update] Process: {process_name}, Type: {app_type}")
        # TODO Phase 2: Gọi AI handler ở đây

    monitor.status_changed.connect(on_status_changed)
    monitor.start()

    window.show()

    # Dừng monitor khi thoát app
    app.aboutToQuit.connect(monitor.stop)

    sys.exit(app.exec())


if __name__ == "__main__":
    main()
```
